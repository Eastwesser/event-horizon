tsx
import { useState, useEffect } from 'react';
import { Balance } from '../components/Billing/Balance';
import api from '../services/api';

interface Item {
  id: string;
  name: string;
  description: string;
  price: number;
  category: string; // 'game_skin', 'merch', 'profile_theme'
  game_id?: string; // 'flappy', 'hexagon', etc.
  image_url: string;
  available: boolean;
}

export function Shop() {
  const [items, setItems] = useState<Item[]>([]);
  const [category, setCategory] = useState('all');
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchItems = async () => {
      try {
        const response = await api.get('/shop/items', {
          params: { category: category !== 'all' ? category : undefined }
        });
        setItems(response.data);
      } catch (err) {
        console.error('Failed to fetch items:', err);
      } finally {
        setLoading(false);
      }
    };
    fetchItems();
  }, [category]);

  const handlePurchase = async (itemId: string) => {
    if (!confirm('Купить этот товар за билетики?')) return;
    try {
      await api.post('/shop/purchase', { item_id: itemId });
      alert('Покупка успешна!');
      // Обновить список товаров (или инвентарь)
    } catch (err) {
      alert('Ошибка при покупке');
    }
  };

  return (
    <div className="shop-container">
      <div className="shop-header">
        <h2>🛒 Магазин</h2>
        <Balance />
      </div>

      <div className="shop-tabs">
        <button onClick={() => setCategory('all')} className={category === 'all' ? 'active' : ''}>Все</button>
        <button onClick={() => setCategory('game_skin')} className={category === 'game_skin' ? 'active' : ''}>Кастомизация</button>
        <button onClick={() => setCategory('merch')} className={category === 'merch' ? 'active' : ''}>Мерч</button>
      </div>

      <div className="shop-items-grid">
        {loading ? (
          <p>Загрузка...</p>
        ) : (
          items.map((item) => (
            <div key={item.id} className="shop-item-card">
              <img src={item.image_url || '/placeholder.png'} alt={item.name} />
              <h4>{item.name}</h4>
              <p>{item.description}</p>
              <p className="price">🎫 {item.price} билетиков</p>
              <button onClick={() => handlePurchase(item.id)} disabled={!item.available}>
                {item.available ? 'Купить' : 'Недоступно'}
              </button>
            </div>
          ))
        )}
      </div>
    </div>
  );
}
