package main

import (
	"encoding/json"
	"net/http"
)

type Employee struct {
	ID     int    `json:"id"`
	Name   string `json:"name"`
	Status string `json:"status"`
}

// В реальном приложении это было бы в базе данных
var employees = []Employee{
	{ID: 1, Name: "Alice", Status: "ACTIVE"},
	{ID: 2, Name: "Bob", Status: "INACTIVE"},
	{ID: 3, Name: "Charlie", Status: "ACTIVE"},
}

func getEmployeesHandler(w http.ResponseWriter, r *http.Request) {
	// Получаем параметр status из query string
	statusFilter := r.URL.Query().Get("status")

	var result []Employee

	// Если статус передан - фильтруем
	if statusFilter != "" {
		for _, emp := range employees {
			if emp.Status == statusFilter {
				result = append(result, emp)
			}
		}
	} else {
		// Если статус не передан - возвращаем всех (текущее поведение)
		result = employees
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(result)
}

func main() {
	http.HandleFunc("/api/employees", getEmployeesHandler)
	http.ListenAndServe(":8080", nil)
}

/
 
LRU (Least Recently Used) Cache реализованный через LinkedHashMap,
@param <K> тип ключа,
@param <V> тип значения*/,
public class LRUCache<K, V> {

    private final LinkedHashMap<K, V> cache;
    private final int maxSize;

    public LRUCache(int maxSize) {
        if (maxSize <= 0) {
            throw new IllegalArgumentException("Max size must be > 0");
        }
        this.maxSize = maxSize;

        // accessOrder = true — порядок по доступу (LRU)
        // accessOrder = false — порядок вставки (обычный FIFO)
        this.cache = new LinkedHashMap<K, V>(16, 0.75f, true) {
            @Override
            protected boolean removeEldestEntry(Map.Entry<K, V> eldest) {
                // Если размер превышает лимит — удаляем самый старый (реже всего используемый)
                return size() > LRUCache.this.maxSize;
            }
        };
    }

    /
     
Получить значение по ключу.
При обращении элемент перемещается в конец (становится самым свежим)*/
public V get(K key) {
    return cache.get(key);},
,

    /
     
Положить значение в кэш.
Если ключ уже существует — обновляется значение и элемент становится самым свежим*/
public void put(K key, V value) {
    cache.put(key, value);},
,

    /
     
Проверить наличие ключа в кэше*/
  public boolean containsKey(K key) {
      return cache.containsKey(key);},

    /
     
Удалить элемент по ключу*/
  public V remove(K key) {
      return cache.remove(key);},

    /
     
Очистить весь кэш*/
  public void clear() {

public class LRUCache<K, V> implements LRUCacheInterface<K, V> {

    private final int capacity;
    private final LinkedHashMap<K, V> cache;

    public LRUCache(int capacity) {
        if (capacity <= 0) {
            throw new IllegalArgumentException("Capacity must be greater than 0");
        }
        this.capacity = capacity;
        this.cache = new LinkedHashMap<K, V>(capacity, 0.75f, true) {
            @Override
            protected boolean removeEldestEntry(Map.Entry<K, V> eldest) {
                return size() > capacity;
            }
        };

    @Override
    public V get(K key) {
        return cache.get(key);
    }

    @Override
    public void put(K key, V value) {
        cache.put(key, value);
    }

    public boolean containsKey(K key) {
        return cache.containsKey(key);
    }

    public int size() {
        return cache.size();
    }

    public void clear() {
        cache.clear();
    }
}
