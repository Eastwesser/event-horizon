План нагрузочного тестирования (после MVP)

Этап	Инструмент	Что тестируем
1	Go producer + consumer	Пропускная способность NATS → PostgreSQL
2	k6 + NATS extension	Реалистичная нагрузка (как в продакшене)
3	Jester	Отказоустойчивость JetStream
Паттерн для consumer (запись в БД)

go
// Буферизированный канал между NATS и PostgreSQL
dbChan := make(chan ScoreEvent, 10000)

// Быстрый ack из NATS
js.Subscribe("game.score.*", func(msg *nats.Msg) {
    var event ScoreEvent
    json.Unmarshal(msg.Data, &event)
    
    select {
    case dbChan <- event:
        // ok
    default:
        log.Printf("⚠️ DB queue full, event dropped")
    }
})

// Медленный batch writer
go func() {
    batch := make([]ScoreEvent, 0, 100)
    ticker := time.NewTicker(1 * time.Second)
    for {
        select {
        case event := <-dbChan:
            batch = append(batch, event)
            if len(batch) >= 100 {
                flushBatch(batch) // bulk insert
                batch = batch[:0]
            }
        case <-ticker.C:
            if len(batch) > 0 {
                flushBatch(batch)
                batch = batch[:0]
            }
        }
    }
}()
